-- Todos
INSERT INTO todos (title, status, user_id) VALUES
('Buy milk',        'active',    1),
('Team meeting',    'active',    1),
('Old task',        'completed', 1),
('Write report',    'active',    2),
('Fix bug',         'completed', 2),
('Read book',       'completed', 2);


-- ============ 1. Count Todos by Status ============
SELECT status, COUNT(*) AS todo_count
FROM todos
GROUP BY status;


-- ============ 2. Count Todos per User ============
SELECT u.name, COUNT(t.id) AS todo_count
FROM users u
LEFT JOIN todos t ON t.user_id = u.id
GROUP BY u.id, u.name;


-- ============ 3. Find Users with No Todos ============
SELECT u.id, u.name, u.email
FROM users u
LEFT JOIN todos t ON t.user_id = u.id
WHERE t.id IS NULL;
